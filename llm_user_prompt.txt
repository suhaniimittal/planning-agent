  Real EMAIL channel backed by SMTP (nodemailer), configured from the admin UI.
When SMTP isn't configured it falls back to log-and-succeed so the delivery
pipeline still runs end-to-end in dev/Docker/smoke without credentials —
matching the previous log-provider behaviour.
**This is where a hub template becomes an email.** Templates are authored as plain
text (a photographer editing `{{galleryTitle}}` copy should not be writing HTML), and
that text used to be passed straight into the `html` field — so every newline
collapsed, the studio's client got one run-on paragraph, and the gallery link arrived
as bare text in the middle of a sentence. `contentFromPlainText` structures it and the
composer renders it under the sending studio's own name and logo.
  ```typescript
class SmtpEmailProvider implements NotificationProvider {
  readonly channel = NotificationChannel.EMAIL;
  readonly name = 'smtp';
  private readonly logger = new Logger('Notify:EMAIL');

  constructor(
    private readonly mailer: MailerService,
    private readonly settings: SmtpSettingsService,
    private readonly email: EmailComposerService,
    private readonly log: EmailLogService,
  ) {}

  get configured(): boolean {
    return this.settings.isConfigured();
  }

  async send(message: OutboundMessage): Promise<ProviderResult> {
    if (!this.settings.isConfigured()) {
      this.logger.debug(
        `[log] → ${message.toAddress ?? '(none)'} :: ${message.subject ?? message.body.slice(0, 60)}`,
      );
      // Record the non-send. This branch short-circuits *before* the mailer, so
      // without this the delivery log would show a studio's account emails failing
      // for want of SMTP while its gallery notifications appeared nowhere at all —
      // the queue reports DELIVERED here on purpose, so the whole lifecycle runs in
      // dev and smoke, and that fiction must not reach the operator's log.
      await this.log.record({
        toAddress: message.toAddress ?? '(no address)',
        subject: message.subject ?? 'GalleryFlow',
        kind: EMAIL_KIND.notification(message.eventKey),
        studioId: message.studioId,
        status: 'FAILED',
        error: 'Email (SMTP) is not configured',
      });
      return { status: 'DELIVERED', providerMessageId: `log_${message.id}` };
    }
    if (!message.toAddress) {
      return { status: 'FAILED', error: 'No destination address for EMAIL' };
    }
    try {
      const subject = message.subject ?? getEvent(message.eventKey ?? '')?.label ?? 'GalleryFlow';
      const brand = await this.email.studioBrand(message.studioId);
      const content = contentFromPlainText(message.body, {
        // The subject already states what happened; repeating it as the card heading
        // is the shape every good transactional email has.
        title: subject,
        actionLabel: emailActionLabel(message.eventKey),
      });
      const { html, text, attachments } = await this.email.compose(content, brand);
      const { messageId } = await this.mailer.send({
        to: message.toAddress,
        subject,
        html,
        text,
        attachments,
        context: {
          kind: EMAIL_KIND.notification(message.eventKey),
          studioId: message.studioId,
        },
      });
      return { status: 'SENT', providerMessageId: messageId };
    } catch (err) {
      return { status: 'FAILED', error: (err as Error).message };
    }
  }
}
  ```

- `apps/api/src/common/core/entitlement-cache.ts:14-68` `apps/api/src/common/core/entitlement-cache.ts::EntitlementCache` — `class EntitlementCache`
  A small TTL cache for hot authorization lookups — feature checks and the
active-subscription resolve run on nearly every protected request, so caching
them avoids a database round-trip each time.
This ships as an in-process Map (correct and dependency-free for a single
node). The public surface (get/set/wrap/invalidate) is intentionally the
shape of a Redis client, so a production multi-node deployment can drop in a
Redis-backed implementation behind the same token without touching callers.
  ```typescript
class EntitlementCache {
  private readonly store = new Map<string, { value: unknown; expiresAt: number }>();

  /** Cache key for a studio's resolved entitlements. */
  static entitlementsKey(studioId: string): string {
    return `entitlements:${studioId}`;
  }

  /** Cache key for a studio's active subscription snapshot. */
  static subscriptionKey(studioId: string): string {
    return `subscription:${studioId}`;
  }

  get<T>(key: string): T | undefined {
    const hit = this.store.get(key);
    if (!hit) {
      return undefined;
    }
    if (hit.expiresAt <= Date.now()) {
      this.store.delete(key);
      return undefined;
    }
    return hit.value as T;
  }

  set<T>(key: string, value: T, ttlMs: number): void {
    this.store.set(key, { value, expiresAt: Date.now() + ttlMs });
  }

  /** Read-through cache: return the cached value or compute, store, and return it. */
  async wrap<T>(key: string, ttlMs: number, compute: () => Promise<T>): Promise<T> {
    const cached = this.get<T>(key);
    if (cached !== undefined) {
      return cached;
    }
    const value = await compute();
    this.set(key, value, ttlMs);
    return value;
  ```

as a query parameter on the photo, render and download routes as well as on

any added later.

gallery for the couple it belongs to, turning a brute-force defence into a
denial-of-service. The trade is that a caller with many addresses gets many
budgets — which is what the IP-keyed route limiter is for, and why both exist.
Entries live in {@link CacheService} for the same reason reset codes do: they
are short-lived counters, worthless once expired, and a durable table would be
pure garbage collection. On a multi-instance deployment this needs Redis to be
shared, exactly like the share-token cache it sits beside.

  ```typescript
class ShareUnlockService {
  private readonly logger = new Logger(ShareUnlockService.name);

  constructor(private readonly cache: CacheService) {}

  private key(token: string, ip?: string): string {
    // An absent IP shouldn't happen (Express populates `req.ip`, and `main.ts`
    // trusts one proxy hop so it is the real caller), but bucket it separately
    // rather than folding it into a shared tally.
    return `share:unlock:${token}:${ip ?? 'unknown'}`;
  }

  /** Throw if this caller is in a cooldown from earlier wrong passcodes. */
  async assertNotLocked(token: string, ip?: string): Promise<void> {
    const entry = await this.cache.get<UnlockAttempts>(this.key(token, ip));
    if (entry && entry.blockedUntil > Date.now()) {
      const minutes = Math.max(1, Math.ceil((entry.blockedUntil - Date.now()) / 60_000));
      throw new ForbiddenException(
        `Too many incorrect passcodes. Try again in ${minutes} minute${minutes === 1 ? '' : 's'}.`,
      );
    }
  }

  /** Count a wrong passcode, starting a cooldown once the allowance is spent. */
  async recordFailure(token: string, ip?: string): Promise<void> {
    const cacheKey = this.key(token, ip);
    const entry = await this.cache.get<UnlockAttempts>(cacheKey);
    const failures = (entry?.failures ?? 0) + 1;
    if (failures >= SHARE_UNLOCK_MAX_FAILURES) {
      this.logger.warn(
        `share ${token.slice(0, 8)}… locked for ${ip ?? 'unknown ip'} after ${failures} wrong passcodes`,
      );
      await this.cache.set<UnlockAttempts>(
        cacheKey,
        { failures, blockedUntil: Date.now() + SHARE_UNLOCK_COOLDOWN_MS },
        SHARE_UNLOCK_COOLDOWN_MS,
      );
      return;
    }
    await this.cache.set<UnlockAttempts>(
      cacheKey,
      { failures, blockedUntil: 0 },
      SHARE_UNLOCK_WINDOW_MS,
    );
  }

  /** Clear the tally — the caller proved they know the passcode. */
  async recordSuccess(token: string, ip?: string): Promise<void> {
    await this.cache.del(this.key(token, ip));
  }
  ```

## Failing closed

`blurDataUrl` placeholder — never to the original bytes. That covers a missing

  ```typescript
class DisplayImageService {
  private readonly logger = new Logger(DisplayImageService.name);
  /** Cached `sharp` factory: `undefined` = not yet tried, `null` = unavailable. */
  private sharpFactory?: SharpFactory | null;
  /** Rendered display copies, keyed by photo + size + watermark fingerprint. */
  private readonly renders = new Map<string, RenderedImage>();
  private renderBytes = 0;
  /** Fetched source bytes, keyed by photo + size. Viewer-independent. */
  private readonly sources = new Map<string, Buffer>();
  private sourceBytes = 0;
  /** Watermark overlay images, inlined and measured, keyed by their source URL. */
  private readonly watermarkImages = new Map<string, WatermarkImage | null>();

  constructor(private readonly storage: StorageService) {}

  /**
   * Produce a display copy of `photo` at `size`, with `watermark` baked in.
   *
   * Results are memoised per (photo, size, watermark) so a gallery of 500 photos
   * costs one render each rather than one per viewer per scroll.
   */
  async render(
    photo: RenderablePhoto,
    size: DisplaySize,
    watermark: WatermarkSpec | null,
  ): Promise<RenderedImage> {
    const cacheKey = `${photo.id}:${size}:${watermarkFingerprint(watermark)}`;
    const hit = this.renders.get(cacheKey);
    if (hit) {
      return hit;
    }
    const rendered = await this.compose(photo, size, watermark);
    // Only cache real renders — a degraded placeholder should be retried once the
    // underlying cause (missing binary, unreachable bucket) is fixed.
    if (!rendered.degraded) {
      this.remember(cacheKey, rendered);
    }
    return rendered;
  }

  /**
   * Fetch and cache a photo's source bytes for a tier without producing a render.
   *
   * This is the half of the work that is the *same for every viewer*, and it is
   * the expensive half: the object-store round trip. Warming it after upload
   * makes the first client's page load fast even when the watermark is
   * per-viewer (`{email}`/`{datetime}`), which the render cache cannot help with
   * because its key includes the mark's fingerprint.
   *
   * **Only warms a photo that already has a derivative.** Without one,
   * `sourceKey` falls back to the full-size original — 10–100× the bytes of a
   * thumb derivative — so warming a large not-yet-processed gallery would pull
   * gigabytes past a 128MB budget, evicting the entries a viewer is actually
   * scrolling through and leaving the cache worse than untouched. The derivative
   * lands within seconds and warming re-runs then, so skipping here costs
   * nothing but a brief cold window.
   *
   * Best-effort and silent otherwise: a failure just means the request path
   * fetches as it always did.
   */
  async prewarmSource(photo: RenderablePhoto, size: DisplaySize): Promise<boolean> {
    if (this.sourceKey(photo, size) === photo.storageKey) {
      return false;
    }
    try {
      return (await this.sourceFor(photo, size)) !== null;
    } catch {
      return false;
    }
  }

  /** Drop every cached render *and* source of one photo (its pixels or mark changed). */
  invalidate(photoId: string): void {
    for (const key of [...this.renders.keys()]) {
      if (key.startsWith(`${photoId}:`)) {
        const entry = this.renders.get(key);
        if (entry) {
          this.renderBytes -= entry.buffer.byteLength;
        }
        this.renders.delete(key);
      }
    }
    for (const key of [...this.sources.keys()]) {
      if (key.startsWith(`${photoId}:`)) {
        const buffer = this.sources.get(key);
        if (buffer) {
          this.sourceBytes -= buffer.byteLength;
        }
        this.sources.delete(key);
      }
    }
  }

  /**
   * The source bytes for a tier, from cache when possible.
   *
   * Caching the *fetched original/derivative* rather than a decoded bitmap keeps
   * the render a single `sharp` pipeline — decoding to raw and re-encoding later
   * would add a lossy round trip, and a raw preview bitmap is ~5MB against
   * ~100KB for the encoded source.
   */
  private async sourceFor(photo: RenderablePhoto, size: DisplaySize): Promise<Buffer | null> {
    const cacheKey = `${photo.id}:${size}`;
    const hit = this.sources.get(cacheKey);
    if (hit) {
      return hit;
    }
    // Server-minted key, except for seeded demo photos which are external URLs.
    const object = await this.storage.fetchObject(this.sourceKey(photo, size), {
      allowAbsolute: true,
    });
    if (!object || object.buffer.byteLength === 0) {
      return null;
    }
    this.sources.set(cacheKey, object.buffer);
    this.sourceBytes += object.buffer.byteLength;
    while (this.sourceBytes > SOURCE_CACHE_MAX_BYTES && this.sources.size > 1) {
      const oldest = this.sources.keys().next().value;
      if (oldest === undefined) break;
      const evicted = this.sources.get(oldest);
      this.sources.delete(oldest);
      if (evicted) {
        this.sourceBytes -= evicted.byteLength;
      }
    }
    return object.buffer;
  }

  /**
   * Which object key to read pixels from for a given tier.
   *
   * Prefers an already-generated derivative — it is smaller, so decoding is
   * cheaper — and falls back to the original *as a render source only*. The
   * original's key is read here and never returned to a caller.
   */
  private sourceKey(photo: RenderablePhoto, size: DisplaySize): string {
    const derived =
      size === 'thumb'
        ? (thumbnailKey(photo.variants) ?? previewKey(photo.variants))
        : previewKey(photo.variants);
    return derived ?? photo.storageKey;
  }

  private async compose(
    photo: RenderablePhoto,
    size: DisplaySize,
    watermark: WatermarkSpec | null,
  ): Promise<RenderedImage> {
    const sharp = await this.loadSharp();
    if (!sharp) {
      return this.degrade(photo, 'sharp is unavailable');
    }
    // Everything below is inside the try: a render must degrade, never throw, so
    // one unreachable object can't take a whole gallery down.
    try {
      // Cached across viewers: the fetch, not the pixels, is what costs.
      const source = await this.sourceFor(photo, size);
      if (!source) {
        return this.degrade(photo, 'source bytes are unavailable');
      }

      const target = DISPLAY_LONG_EDGE[size];
      const { width, height } = await this.fittedSize(sharp, source, target);

      // `.rotate()` bakes EXIF orientation before metadata is dropped, so the copy
      // is upright even though it carries no orientation tag. Resize, composite
      // and encode in one pipeline — laying the mark on an already-encoded JPEG
      // would cost a second lossy round-trip on every photo.
      const pipeline = sharp(source, { failOn: 'none' })
        .rotate()
        .resize({ width: target, height: target, fit: 'inside', withoutEnlargement: true });

      const overlay = await this.overlayFor(watermark, width, height, size);
      if (overlay) {
        pipeline.composite([{ input: Buffer.from(overlay), top: 0, left: 0 }]);
      }
      const buffer = await pipeline.jpeg({ quality: DISPLAY_QUALITY, mozjpeg: true }).toBuffer();
      return { buffer, contentType: 'image/jpeg', degraded: false };
    } catch (error) {
      // A watermark that fails to rasterise (missing fontconfig, malformed
      // template) must not hand back the original — degrade instead.
      return this.degrade(photo, `render failed: ${(error as Error).message}`);
    }
  }

  /**
   * The dimensions the resize will actually produce, derived from the source's
   * metadata rather than by encoding twice.
   *
   * Deliberately floors: `sharp` rounds, and `floor <= round`, so the overlay is
   * never *larger* than the image it composites onto — which `sharp` rejects
   * outright. A sub-pixel difference in mark placement is invisible; a thrown
   * composite would cost the watermark entirely.
   */
  private async fittedSize(
    sharp: SharpFactory,
    input: Buffer,
    target: number,
  ): Promise<{ width: number; height: number }> {
    const meta = await sharp(input).metadata();
    // EXIF orientations 5–8 are the quarter-turns; `.rotate()` will swap the axes,
    // so the post-rotation extent is what the mark must be laid out against.
    const turned = (meta.orientation ?? 1) >= 5;
    const width = (turned ? meta.height : meta.width) ?? 0;
    const height = (turned ? meta.width : meta.height) ?? 0;
    if (width <= 0 || height <= 0) {
      throw new Error('source has no readable dimensions');
    }
    const scale = Math.min(1, target / Math.max(width, height));
    return {
      width: Math.max(1, Math.floor(width * scale)),
      height: Math.max(1, Math.floor(height * scale)),
    };
  }

  /** The SVG overlay bytes for a spec at a given size, or null when there is none. */
  private async overlayFor(
    watermark: WatermarkSpec | null,
    width: number,
    height: number,
    size: DisplaySize,
  ): Promise<string | null> {
    if (!watermark?.enabled) {
      return null;
    }
    const image = watermark.imageUrl ? await this.watermarkImage(watermark.imageUrl) : null;
    return buildWatermarkSvg(watermark, width, height, {
      image,
      // Thumbnails are shown in square tiles with `object-cover`, so a mark in the
      // true corner is cropped off screen — visible in the bytes, invisible to the
      // viewer it is meant to deter.
      cropSafe: size === 'thumb',
    });
  }

  /**
   * Fetch a watermark overlay image once and hold it inlined and measured.
   *
   * librsvg will not resolve remote `href`s, so the image has to be inlined — and
   * it has to be *measured*, because an `<image>` without an explicit height is
   * drawn at its intrinsic size, which ignores the studio's `scale` entirely. A
   * studio has a handful of templates at most, so a small map is plenty.
   */
  private async watermarkImage(url: string): Promise<WatermarkImage | null> {
    if (this.watermarkImages.has(url)) {
      return this.watermarkImages.get(url) ?? null;
    }
    const image = await this.loadWatermarkImage(url);
    if (this.watermarkImages.size >= WATERMARK_IMAGE_CACHE_MAX) {
      const oldest = this.watermarkImages.keys().next().value;
      if (oldest !== undefined) {
        this.watermarkImages.delete(oldest);
      }
    }
    this.watermarkImages.set(url, image);
    return image;
  }

  /** Read a logo's bytes and aspect ratio, or null if either is unavailable. */
  private async loadWatermarkImage(url: string): Promise<WatermarkImage | null> {
    const sharp = await this.loadSharp();
    if (!sharp) {
      return null;
    }
    try {
      const object = await this.storage.fetchObject(url);
      if (!object || object.buffer.byteLength === 0) {
        return null;
      }
      const meta = await sharp(object.buffer).metadata();
      if (!meta.width || !meta.height) {
        return null;
      }
      // The mime comes from what `sharp` actually decoded, never from the response
      // header — that string is interpolated into SVG markup, and a remote server
      // does not get to choose what goes into our document. An unsupported format
      // drops the image half of the mark rather than declaring a mime we'd be
      // guessing at.
      const mime = IMAGE_MIME[meta.format ?? ''];
      if (!mime) {
        return null;
      }
      return {
        uri: `data:${mime};base64,${object.buffer.toString('base64')}`,
        aspect: meta.width / meta.height,
      };
    } catch (error) {
      // An unreadable logo costs the image half of the mark, not the whole render:
      // the text half still composites and the photo still fails closed.
      this.logger.warn(`watermark image ${url} is unusable: ${(error as Error).message}`);
      return null;
    }
  }

  /**
   * The fail-closed response: something safe and visibly not the photograph.
   *
   * Prefers the photo's own `blurDataUrl` — a handful of pixels generated
   * client-side at upload, conveying colour and nothing else. Failing that it
   * serves {@link PLACEHOLDER_SVG}.
   *
   * This must never throw. It used to raise a 503 when `blurDataUrl` was absent,
   * which is the case for every photo predating LQIP capture (half the seeded
   * demo set) — so one transient fetch failure turned a gallery into a wall of
   * broken images with no indication of why. Degrading is the whole point; it has
   * to degrade for *every* photo, not just recently uploaded ones.
   */
  private degrade(photo: RenderablePhoto, reason: string): RenderedImage {
    this.logger.warn(`photo ${photo.id}: serving placeholder — ${reason}`);
    const decoded = decodeDataUri(photo.blurDataUrl);
    if (decoded) {
      return { ...decoded, degraded: true };
    }
    return {
      buffer: Buffer.from(PLACEHOLDER_SVG, 'utf8'),
      // An SVG needs no image encoder, so this path stays available even when
      // `sharp` is the thing that failed — and it scales to any tile size.
      contentType: 'image/svg+xml',
      degraded: true,
    };
  }

  /** Insert a render, evicting oldest-first to stay inside the byte budget. */
  private remember(key: string, rendered: RenderedImage): void {
    this.renders.set(key, rendered);
    this.renderBytes += rendered.buffer.byteLength;
    while (this.renderBytes > RENDER_CACHE_MAX_BYTES && this.renders.size > 1) {
      const oldest = this.renders.keys().next().value;
      if (oldest === undefined) break;
      const entry = this.renders.get(oldest);
      this.renders.delete(oldest);
      if (entry) {
        this.renderBytes -= entry.buffer.byteLength;
      }
    }
  }

  /**
   * Resolve the `sharp` factory once, defensively — mirroring
   * `PhotoProcessingService`. Unlike there, an unavailable binary here does not
   * mean "leave the original serving": it means every display request degrades to
   * a placeholder, which is loud enough to get noticed.
   */
  private async loadSharp(): Promise<SharpFactory | null> {
    if (this.sharpFactory !== undefined) {
      return this.sharpFactory;
    }
    try {
      const mod = await import('sharp');
      this.sharpFactory = mod.default ?? (mod as unknown as SharpFactory);
    } catch (error) {
      this.logger.error(
        `sharp is unavailable — protected image rendering is disabled, galleries will ` +
          `show placeholders: ${(error as Error).message}`,
      );
      this.sharpFactory = null;
    }
    return this.sharpFactory;
  }
}
  ```
