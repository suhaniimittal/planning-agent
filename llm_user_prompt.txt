## Issue

First, in the payment system, we don't currently have a way to specify a currency alongside a payment amount — we'd like to add a currency field to payment requests and validate that the given currency code is actually supported before processing the payment. Second, and unrelated to the first, bank transfers have two problems: withdrawing more money than an account's current balance is allowed to go through instead of being rejected, and transfers of zero or negative amounts are also allowed instead of being rejected.

  ```python
class Cart:
    """Tracks items and quantities for a single shopping session."""

    def __init__(self) -> None:
        # Maps Item -> quantity. Adding the same item twice merges quantities
        # rather than creating a duplicate line.
        self._lines: dict[Item, int] = {}

    def add_item(self, item: Item, quantity: int = 1) -> None:
        """Add `quantity` units of `item` to the cart.

        If the item is already in the cart, the quantities are merged
        instead of creating a second line for the same item.
        """
        if quantity <= 0:
            raise ValueError(f"quantity must be positive, got {quantity}")

        self._lines[item] = self._lines.get(item, 0) + quantity

    def remove_item(self, item: Item) -> None:
        """Remove an item (and all of its quantity) from the cart entirely."""
        self._lines.pop(item, None)

    def quantity_of(self, item: Item) -> int:
        """Return how many units of `item` are currently in the cart."""
        return self._lines.get(item, 0)

    def line_count(self) -> int:
        """Return the number of distinct items (lines) in the cart."""
        return len(self._lines)

    def _line_subtotal(self, item: Item, quantity: int) -> float:
        subtotal = item.unit_price * quantity
        # BUG: buying exactly BULK_DISCOUNT_THRESHOLD units should qualify
        # for the discount ("10 or more"), but `>` excludes the boundary
        # case, so a cart with exactly 10 units gets no discount at all.
        if quantity > BULK_DISCOUNT_THRESHOLD:
            subtotal *= 1 - BULK_DISCOUNT_RATE
        return subtotal

    def subtotal(self) -> float:
        """Return the cart total, after any per-line bulk discounts."""
        return sum(
            self._line_subtotal(item, quantity)
            for item, quantity in self._lines.items()
        )
  ```

  ```java
public class TransferService {

    private final List<Transaction> history = new ArrayList<>();

    /**
     * Move {@code amount} from {@code from} to {@code to}.
     *
     * BUG: there's no check here that {@code amount} is actually positive.
     * A caller can pass zero or a negative number straight through to
     * withdraw/deposit below, which (combined with Account's own missing
     * balance check) lets a "transfer" invent money out of nowhere instead
     * of just moving it.
     */
    public Transaction transfer(Account from, Account to, double amount) {
        from.withdraw(amount);
        to.deposit(amount);

        Transaction transaction = new Transaction(from.getAccountId(), to.getAccountId(), amount);
        history.add(transaction);
        return transaction;
    }

    public List<Transaction> getHistory() {
        return history;
    }
}
  ```
